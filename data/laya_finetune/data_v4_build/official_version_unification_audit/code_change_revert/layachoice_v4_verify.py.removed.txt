#!/usr/bin/env python3
"""LayaChoice V4 dynamic-option bundle verifier.

V4 keeps the frozen LayaChoice-v2 row contract and changes exactly one thing:
the option count is the real K capability cards + one REJECT (1..4 options),
instead of a fixed 3 + 1.

Runs with the standard library only (no torch / no tokenizer / no model, no
network), reusing the same adapter the training path uses:

    python layachoice_v4_verify.py --data DATA_DIR

Checks:
  * manifest identity: bench, B_noprov, K+1 option slots, 14 frozen row keys;
  * recorded source SHA256 values against the files on disk;
  * every split loads through the training adapter, counts match the manifest;
  * per-row option count, indices, REJECT semantics, and shuffle reproduction;
  * in-place K=0->K=1 repaired rows match the manifest ``repair`` block;
  * calibration indices/ids reproduce, splits are disjoint;
  * gold slot distribution matches the manifest;
  * the construction audit covers every row;
  * the bundle is not an archived/superseded copy.

Exits non-zero on the first class of failure found; prints a PASS/FAIL summary.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import random
import sys
from collections import Counter
from pathlib import Path

HERE = Path(__file__).resolve().parent
if str(HERE) not in sys.path:
    sys.path.insert(0, str(HERE))

import layachoice_v2_dataset as D  # noqa: E402
import layachoice_v2_spec as S     # noqa: E402

SPLITS = ("train", "val", "test")
REJECT = S.REJECT_LABEL
REPAIR_PREFIX = "k0-to-k1-inplace-v1"
DEFAULT_PREFIX = "reject-shuffle-v1"

_failures: list[str] = []


def check(name: str, ok: bool, detail: str = "") -> bool:
    mark = "ok  " if ok else "FAIL"
    print(f"  [{mark}] {name}" + (f" — {detail}" if detail else ""))
    if not ok:
        _failures.append(name)
    return ok


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def raw_rows(data: Path, split: str) -> list[dict]:
    path = data / f"v2_{split}.jsonl"
    with open(path, encoding="utf-8") as fh:
        return [json.loads(line) for line in fh if line.strip()]


def reproduced_perm(prefix: str, row_id: str, n: int) -> list[int]:
    perm = list(range(n))
    random.Random(f"{prefix}:{row_id}").shuffle(perm)
    return perm


def verify_identity(manifest: dict) -> None:
    print("[1] manifest identity")
    check("bench is LayaChoice-v2", manifest.get("bench") == S.V2_BENCH,
          str(manifest.get("bench")))
    slots = manifest.get("option_slots")
    check("option_slots is the K+1 contract",
          isinstance(slots, dict) and slots.get("rule") == "K+1"
          and slots.get("min") == 1 and slots.get("max") == 4,
          str(slots))
    check("formal card view is B_noprov",
          manifest.get("formal_card_view") == "B_noprov",
          str(manifest.get("formal_card_view")))
    check("training schema keys are the frozen V2 14 keys",
          set(manifest.get("training_schema_keys", [])) == set(D.ROW_KEYS),
          str(sorted(set(D.ROW_KEYS) ^ set(manifest.get("training_schema_keys", [])))))
    check("base revision matches spec",
          manifest.get("base_checkpoint", {}).get("revision") == S.BASE_REVISION,
          str(manifest.get("base_checkpoint", {}).get("revision")))


def verify_hashes(data: Path, manifest: dict) -> None:
    print("[2] recorded source hashes")
    for name, want in manifest.get("source_files", {}).items():
        if name == "source_csv":
            check("source_csv recorded (upstream input, not bundled)", bool(want))
            continue
        path = data / name
        if not path.is_file():
            check(f"{name} present", False, "recorded but missing")
            continue
        got = sha256(path)
        check(f"{name} sha256", got == want, got)


def verify_rows(data: Path, manifest: dict) -> dict[str, list[dict]]:
    print("[3] rows, dynamic option counts, and labels")
    reject_card = manifest["reject"]["card"]
    loaded: dict[str, list[dict]] = {}
    k_observed: set[int] = set()
    repair_counts: Counter = Counter()
    ids: dict[str, set[str]] = {}
    query_ids: dict[str, set[str]] = {}

    for split in SPLITS:
        rows = raw_rows(data, split)
        loaded[split] = rows
        D.load_split(split, data)  # adapter checks schema, indices, permutation
        check(f"{split}: row count == manifest {manifest['rows'][split]}",
              len(rows) == manifest["rows"][split], str(len(rows)))
        ids[split] = set()
        query_ids[split] = set()
        for row in rows:
            rid = row["id"]
            n = len(row["order"])
            k_observed.add(n - 1)
            check_pair = (
                len(row["options"]) == n,
                row["order"].count(REJECT) == 1,
                row["reject_index"] == row["order"].index(REJECT),
                row["shuffle"]["permutation"] == reproduced_perm(
                    row["shuffle"]["seed_prefix"], rid, n),
                all(text == reject_card for cid, text in
                    zip(row["order"], row["options"]) if cid == REJECT),
            )
            if not all(check_pair):
                check(f"{split}/{rid}: row invariants", False, str(check_pair))
                continue
            if row["target_kind"] == "reject":
                ok = (row["gold_capability_id"] is None
                      and row["gold_index"] == row["reject_index"])
            else:
                ok = (row["gold_capability_id"] is not None
                      and row["order"][row["gold_index"]] == row["gold_capability_id"]
                      and row["gold_index"] != row["reject_index"])
            if not ok:
                check(f"{split}/{rid}: label semantics", False)
            prefix = row["shuffle"]["seed_prefix"]
            if prefix.startswith(REPAIR_PREFIX):
                repair_counts[split] += 1
                if len(row["order"]) != 2 or row["target_kind"] != "capability":
                    check(f"{split}/{rid}: repaired row must be K=1", False)
            elif prefix != DEFAULT_PREFIX:
                check(f"{split}/{rid}: unknown shuffle prefix", False, prefix)
            ids[split].add(rid)
            query_ids[split].add(row["source_query_id"])

        check(f"{split}: ids unique", len(ids[split]) == len(rows))
        check(f"{split}: source_query_id unique",
              len(query_ids[split]) == len(rows))

    check(f"K values stay in 0..3", k_observed <= {0, 1, 2, 3}, str(sorted(k_observed)))
    check("manifest option_slots.observed matches the bundle",
          sorted(manifest["option_slots"].get("observed", []))
          == sorted({k + 1 for k in k_observed}),
          f"{manifest['option_slots'].get('observed')} vs {sorted(k + 1 for k in k_observed)}")

    for a in range(len(SPLITS)):
        for b in range(a + 1, len(SPLITS)):
            sa, sb = SPLITS[a], SPLITS[b]
            check(f"{sa}/{sb}: disjoint ids", ids[sa].isdisjoint(ids[sb]))
            check(f"{sa}/{sb}: disjoint source_query_id",
                  query_ids[sa].isdisjoint(query_ids[sb]))

    repair = manifest.get("repair")
    if repair:
        expect = {s: int(repair["repaired_by_split"][s]) for s in SPLITS}
        got = {s: repair_counts.get(s, 0) for s in SPLITS}
        check("repair block counts == rows with the in-place seed prefix",
              got == expect, f"{got} vs {expect}")
        check("repair total == sum(by split)",
              int(repair["repaired_rows"]) == sum(expect.values()),
              str(repair["repaired_rows"]))
        check("repair seed prefix", str(repair.get("seed_prefix", "")).startswith(
            REPAIR_PREFIX), str(repair.get("seed_prefix")))
    else:
        check("no repair rows without a repair block",
              sum(repair_counts.values()) == 0, str(sum(repair_counts.values())))

    for split in SPLITS:
        want = {k: v for k, v in sorted(manifest["gold_slot_distribution"][split].items())}
        got = {str(k): v for k, v in sorted(
            Counter(str(r["gold_index"]) for r in loaded[split]).items())}
        check(f"{split}: gold slot distribution matches", got == want, f"{got} vs {want}")
    return loaded


def verify_calibration(data: Path, manifest: dict) -> None:
    print("[4] calibration split")
    train = D.load_split("train", data)
    calib, fit = D.frozen_calibration(train, manifest)
    check("calibration rows == manifest",
          len(calib) == manifest["calibration"]["calibration_rows"], str(len(calib)))
    check("effective train rows == manifest",
          len(fit) == manifest["calibration"]["effective_train_rows"], str(len(fit)))
    check("calibration and effective train are disjoint",
          {e.id for e in calib}.isdisjoint({e.id for e in fit}))


def verify_audit(data: Path, manifest: dict) -> None:
    print("[5] construction audit coverage")
    path = data / manifest.get("construction_audit_file", "construction_audit.jsonl")
    check("construction audit exists", path.is_file(), str(path))
    if not path.is_file():
        return
    audit_ids = {r["id"] for r in
                 (json.loads(line) for line in path.open(encoding="utf-8") if line.strip())}
    row_ids = {r["id"] for split in SPLITS for r in raw_rows(data, split)}
    check("construction audit covers exactly the bundle rows", audit_ids == row_ids,
          f"missing={len(row_ids - audit_ids)} extra={len(audit_ids - row_ids)}")


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data", required=True, type=Path)
    args = parser.parse_args()
    data = args.data.resolve()

    print(f"V4 bundle: {data}")
    resolved = str(data).replace("\\", "/").lower()
    if "full_r1_repaired" in resolved or "/archive/" in resolved:
        raise SystemExit("refusing an archived/superseded bundle")

    manifest = D.read_manifest(data)
    verify_identity(manifest)
    verify_hashes(data, manifest)
    verify_rows(data, manifest)
    verify_calibration(data, manifest)
    verify_audit(data, manifest)

    print()
    if _failures:
        print(f"FAIL: {len(_failures)} check(s) failed")
        return 1
    print("PASS: V4 bundle contract verified")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
